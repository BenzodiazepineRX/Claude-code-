def bp_an(z,lo,hi,o=4): return sosfiltfilt(butter(o,[lo,hi],'band',fs=SR,output='sos'),z)
def env_ms(z,ms=10):

def movavg(x,w):

    lo=np.clip(i-w//2,0,len(x)); hi=np.clip(i+w//2+1,0,len(x)); return (c[hi]-c[lo])/(hi-lo)

# ---------- P-center ----------
def pcenter(m, ws, we, hop_ms=2.0):
    """Stressed-vowel P-center of the word in samples [ws,we): envelope = 300 Hz-3 kHz band, RMS on hop_ms hops smoothed 10 ms;
    stressed vowel = envelope max inside the word; P-center = last time before that max where the envelope rises through
    50% (-6 dB) of the max (half-rise point of the vowel). Returns global sample index."""
    pad=int(0.05*SR); a=max(0,ws-pad); b=min(len(m),we+pad)
    z=bp_an(m[a:b],300,3000); h=int(SR*hop_ms/1000)
    n=len(z)//h; e=np.sqrt((z[:n*h].reshape(n,h)**2).mean(1)); e=movavg(e,max(1,int(10/hop_ms)))
    i0=(ws-a)//h; i1=min(n,(we-a)//h)
    if i1<=i0+1: return ws
    ipk=i0+int(np.argmax(e[i0:i1])); half=0.5*e[ipk]; j=ipk
    while j>i0 and e[j-1]>=half: j-=1
    # interpolate crossing between j-1 and j
    if j>0 and e[j]>e[j-1]:
        frac=(half-e[j-1])/(e[j]-e[j-1]); t=(j-1+frac)
    else: t=j
    return int(a+t*h+h/2)

def lag_xcorr(ref, x, maxlag):
    """lag (samples) of x relative to ref, band 1-6 kHz, FFT xcorr, |lag|<=maxlag"""
    r=bp_an(ref,1000,6000); y=bp_an(x,1000,6000)
    n=1<<int(np.ceil(np.log2(len(r)+len(y))))
    c=np.fft.irfft(np.fft.rfft(y,n)*np.conj(np.fft.rfft(r,n)),n)
    c=np.concatenate([c[-maxlag:],c[:maxlag+1]]); return int(np.argmax(c))-maxlag

def pcenter_stressed(m, ws, we, hop_ms=2.0, within_db=3.0):
    """P-center of the lexically stressed vowel, approximated as the FIRST vowel whose peak is within `within_db` of the word's
    maximum (so 'e-NOUGH' -> 'nough', 'AD-der-all' -> 'Ad', where 'rall' is louder by <3 dB); P-center = half-rise (50% amplitude)
    point before that vowel's peak. Envelope: 300 Hz-3 kHz, RMS on 2 ms hops, smoothed 10 ms."""
    pad=int(0.05*SR); a=max(0,ws-pad); b=min(len(m),we+pad)
    z=bp_an(m[a:b],300,3000); h=int(SR*hop_ms/1000)
    n=len(z)//h; e=np.sqrt((z[:n*h].reshape(n,h)**2).mean(1)); e=movavg(e,max(1,int(10/hop_ms)))
    i0=(ws-a)//h; i1=min(n,(we-a)//h); M=e[i0:i1].max(); thr=M*10**(-within_db/20)
    i=i0+int(np.argmax(e[i0:i1]>=thr))
    while i+1<i1 and e[i+1]>=e[i]: i+=1          # climb to that vowel's local peak
    half=0.5*e[i]; j=i
    while j>i0 and e[j-1]>=half: j-=1
    t=(j-1+(half-e[j-1])/(e[j]-e[j-1])) if (j>0 and e[j]>e[j-1]) else j
    return int(a+t*h+h/2)

##### FILE: metrics_g2.py #####
import sys; sys.path.insert(0,'.'); from g2lib import *
def fr(z,ms=10):
    e,h=env_ms(z,ms); return e
def dry_to_wet_db(dry,wet,act,lo=None,hi=None):
    """median over active 10 ms frames of 20log10(dry_rms/wet_rms) (energy-ratio on the pooled active frames)"""
    if lo: dry=bp_an(dry,lo,hi); wet=bp_an(wet,lo,hi)
    ed=fr(dry); ew=fr(wet); n=min(len(ed),len(ew),len(act)); a=act[:n]
    if a.sum()==0: return float('nan')
    return float(pdb((ed[:n][a]**2).sum())-pdb((ew[:n][a]**2).sum()))
def under_db(lead,layer,act,lo,hi): return dry_to_wet_db(lead,layer,act,lo,hi)
def tail_end_dbfs(y,ms=200):
    y=y if y.ndim==1 else y.max(1) if False else np.abs(y).max(1)
    n=int(SR*ms/1000); return float(db(np.sqrt(np.mean(y[-n:]**2))))
def corr(x):
    L=x[:,0]-x[:,0].mean(); R=x[:,1]-x[:,1].mean(); return float((L*R).sum()/np.sqrt((L*L).sum()*(R*R).sum()+1e-30))
